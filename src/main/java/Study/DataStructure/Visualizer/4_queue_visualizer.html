큐(Queue)

FIFO - 배열 기반(고정 -> 동적 -> 원형) 3단계 개선 + 연결리스트 기반 구현

📖 큐(Queue) 학습 노트 (클릭해서 펼치기)

📥 Queue 개요

  • FIFO(First In First Out) - 먼저 삽입된 원소가 가장 먼저 출력 (Stack의 LIFO와 대칭)
  • front: 다음에 나갈(dequeue) 원소의 위치, rear: 다음에 들어갈(enqueue) 자리
  • enqueue / dequeue / peek 모두 이상적으로 O(1), 배열 기반은 resize 시 Amortized O(1)

🧱 고정 배열 선형 큐

  • front/rear 인덱스만 증가시키는 단순한 구현
  • 한계: rear가 배열 끝(capacity)에 도달하면, front 쪽에 빈 공간이 남아있어도 더 이상 enqueue 불가(메모리 낭비)

🔁 동적 배열 선형 큐

  • rear==capacity 이면서 front>0(앞쪽 빈 공간 존재) -> relocation(앞으로 당김, O(n))
  • 진짜 꽉 찼으면(size == capacity) -> resizing(용량 2배, O(n))
  • relocation이 매번 발생하진 않지만, 발생 시점마다 O(n) 물리적 이동 비용 존재 - 원형 큐로 가기 전 중간 단계

🌀 동적 배열 원형 큐 (핵심)

  • index를 (index % capacity)로 순환(rotate)시켜 배열의 물리적 끝과 논리적 끝을 분리
  • 진짜 꽉 찼을 때만 index rotate 수행
  • front/rear가 배열 끝에 도달해도 공간이 있으면 0번 인덱스부터 재사용 - relocation 자체가 불필요
  • 진짜 꽉 찼을 때만 resize(용량 2배) -> Dynamic Array와 동일한 Amortized O(1) 근거
  • front==rear가 "꽉 참"과 "빔" 두 상태 모두에 해당할 수 있음 -> size 필드로 명시적으로 구분 (다른 방법: 한 칸을 일부러 비워두는 방식도 존재)

🔗 연결리스트 기반 큐

  • head = front, tail = rear
  • enqueue: tail.next = newNode, tail = newNode (O(1)) / dequeue: head = head.next (O(1))
  • 단일 연결리스트 + tail 포인터만으로 충분 - dequeue는 항상 head 방향으로만 이동하므로 prev가 필요 없음 (이중 연결 불필요)
  • head가 null이 되면(큐가 빔) tail도 반드시 null 처리 - 안 하면 tail이 유령 노드를 참조

⚖️ 배열(원형) vs 연결리스트 선택

  • 배열(원형 큐): 캐시 지역성 좋음, 포인터 오버헤드 없음, 단 resize 시점에 O(n) 스파이크 존재
  • 연결리스트: 항상 균일한 O(1)(스파이크 없음), 재할당 없이 동적 확장 가능, 대신 노드당 포인터 메모리 오버헤드

⏱ 시간/공간 복잡도 비교

① 고정 배열 선형 큐 
// rear가 배열 끝에 도달하면 front에 공간이 남아있어도 enqueue 불가 (메모리 낭비)
public class FixedArrayBasedLinearQueue<E> {

  private final Object[] queue;
  private int front = 0;
  private int rear = 0;
  private int size = 0;
  private final int capacity;

  public FixedArrayBasedLinearQueue() {
    this.capacity = 10;
    this.queue = new Object[capacity];
  }

  public FixedArrayBasedLinearQueue(int capacity) {
    if (capacity <= 0) {
          throw new IllegalArgumentException
          ("잘못된 용량 초기화 capacity: " + capacity);
    }
    this.capacity = capacity;
    this.queue = new Object[capacity];
  }

  public void enqueue(E e) {
    checkEnqueueable();

    queue[rear] = e;
    rear++;
    size++;
  }

  private void checkEnqueueable() {
    if (isFull()) {throw new IllegalStateException("큐가 가득 참");}

    if (rear == capacity) {
      throw new IllegalStateException("마지막 요소가 큐의 끝에 도달함");
    }
  }
}
② 동적 배열 선형 큐 
// relocation으로 앞쪽 빈 공간 재사용, 진짜 꽉 찼을 때만 resize
public class DynamicArrayBasedLinearQueue<E> {

  private Object[] queue;
  private int front = 0;
  private int rear = 0;
  private int size = 0;
  private int capacity;

  public DynamicArrayBasedLinearQueue() {
    this.capacity = 10;
    this.queue = new Object[capacity];
  }

  public DynamicArrayBasedLinearQueue(int capacity) {
    if (capacity <= 0) {
          throw new IllegalArgumentException
          ("잘못된 용량 초기화 capacity: " + capacity);
    }
    this.capacity = capacity;
    this.queue = new Object[capacity];
  }

  public void enqueue(E e) {
    doRelocationOrResizingIfNeeded();

    queue[rear] = e;
    rear++;
    size++;
  }

  private void doRelocationOrResizingIfNeeded() {
    if ((rear == capacity && front > 0)) {
      relocation();
    } else if (isFull()) {
      resizing();
    }
  }

  private void resizing() {
    capacity *= 2;
    queue = Arrays.copyOf(queue, capacity);
  }

  private void relocation() {
    System.arraycopy(queue, front, queue, 0, size);
    front = 0;
    rear = size;
  }
}
③ 동적 배열 원형 큐 (최종 개선판) 
// index를 capacity로 modulo하여 relocation 없이 앞쪽 빈 공간 즉시 재활용
public class DynamicArrayBasedCircularQueue<E> {

  private Object[] queue;
  private int front = 0, rear = 0, size = 0, capacity;

  public DynamicArrayBasedCircularQueue() {
    this.capacity = 10;
    this.queue = new Object[capacity];
  }

  public DynamicArrayBasedCircularQueue(int capacity) {
    if (capacity <= 0) {
          throw new IllegalArgumentException
          ("잘못된 용량 초기화 capacity: " + capacity);
    }
    this.capacity = capacity;
    this.queue = new Object[capacity];
  }

  public void enqueue(E e) {
    doResizingIfNeeded();

    queue[rear] = e;

    rear++;
    size++;

    if (rear == capacity) {rear = rotateIndex(rear);}
  }

  @SuppressWarnings("unchecked")
  public E dequeue() {
    checkQueueSize();
    E value = (E) queue[front];
    queue[front] = null;

    size--;
    front++;

    if (front == capacity) {front = rotateIndex(front);}

    return value;
  }

  public int size() {return size;}

  public int getCapacity() {return capacity;}

  private int rotateIndex(int value) {return value % capacity;}

  private void doResizingIfNeeded() {
    if (isFull()) {
      int prevCapacity = capacity;
      capacity *= 2;
      Object[] newQueue = new Object[capacity];

      for (int i = 0; i < size; i++) {
        newQueue[i] = queue[(front + i) % prevCapacity];
      }

      queue = newQueue;
      front = 0;
      rear = size;
    }
  }
}
④ 연결리스트 기반 큐 
// head(front) / tail(rear) 포인터로 양쪽 끝 O(1) 접근
public class LinkedListBasedQueue<E> {

  private Node<E> head;
  private Node<E> tail;
  private int size;

  static class Node<E> {

    private Node<E> next;
    private E data;

    public Node(E data) {
      this.data = data;
      this.next = null;
    }
  }

  public void enqueue(E e) {
    Node<E> newNode = new Node<>(e);

    if (head == null) {
      head = newNode;
    } else {
      tail.next = newNode;
    }

    tail = newNode;
    size++;
  }

  public E dequeue() {
    checkQueueSize();

    Node<E> output = head;
    head = head.next;
    size--;

    // 큐가 비었을 경우 tail 처리
    if (head == null) {tail = null;}

    return output.data;
  }

  public E peek() {
    checkQueueSize();
    return head.data;
  }

  public int size() {return size;}
}
